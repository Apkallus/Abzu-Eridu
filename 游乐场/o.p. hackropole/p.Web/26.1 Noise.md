## 描述

Noise 是全新的 100% 安全聊天应用。得益于其 start-to-start 隧道，你的消息经过端到端加密。你甚至可以与朋友分享秘密。是不是很棒？

此挑战还有一个更复杂的变体，可在此处获取：[`Noise (harder)`](https://hackropole.fr/en/challenges/web/fcsc2024-web-noise-harder/)。

```yaml
services:
  noise:
    image: anssi/fcsc2024-web-noise:latest
    ports:
      - "8000:3000"

```

挑战提供文件，代码审计型。保存到 [代码审计](#代码审计)

```ts
import puppeteer from "puppeteer";

const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));

// 追踪 flag
const FLAG = "FCSC{flag_placeholder}";
// flag 位于 adminMessages[2]
const adminMessages = [
  "Hello, I'm the admin of this chat. I'm here to help you if you have any questions.",
  "I'm sorry, I can't help you with that.",
  "You can have this flag if you want: " + FLAG,
];

async function visit(roomId: string) {
  let browser;
  try {
    browser = await puppeteer.launch({
      headless: true,
      executablePath: "/usr/bin/chromium-browser",
      args: [
        "--incognito",
        "--no-sandbox",
        "--disable-setuid-sandbox",
        "--disable-gpu",
        "--disable-jit",
        "--disable-wasm",
        "--disable-dev-shm-usage",
        "--ignore-certificate-errors",
        "--js-flags=--noexpose_wasm,--jitless",
      ],
    });

    const page = await browser.newPage();

    // 打开 chat room 端点
    await page.goto(`http://127.0.0.1:3000/chat/${roomId}`);
    await page.waitForSelector("#loaded", {
      timeout: 3000,
    });

    for (const message of adminMessages) {
      await page.focus("#form-input");
      await page.keyboard.type(message);
      await sleep(200);
      await page.click("#form-btn");
      await sleep(2000);
    }
  } finally {
    await browser?.close();
  }
}

async function run(roomId: string | null) {
  if (!roomId?.match(/^[a-f0-9]{32}$/)) {
    return "Invalid room id!";
  }
  try {
    await visit(roomId);
    return `Visited room ${roomId}!`;
  } catch (err) {
    return "Something went wrong!";
  }
}

Bun.serve({
  port: 4242,
  hostname: "127.0.0.1",
  async fetch(req) {
    const url = new URL(req.url);
    const result = await run(url.searchParams.get("roomId"));
    return new Response(result);
  },
});
```

### src\app\composables\useNoise.ts

```ts
export default function useNoise() {
  return {
    encrypt(data: string) {
        // 使用随机值，无法解密
      const key = crypto.getRandomValues(new Uint8Array(data.length));
      let out = "";

      for (let i = 0; i < data.length; i++) {
        out += (data.charCodeAt(i) ^ key[i]).toString(16).padStart(2, "0");
      }
      return out;
    },
  };
}
```
