    ```js
        coins: number;

        let sessionId = getCookie(event, COOKIE_NAME);
        if (!sessionId) {
    ```

    ```js
    // 设置类型
    export interface LootBox {
        id: string;
        seed: number;
    }

    // 创建随机盒子
    /// id 使用2轮随机数转为字符串，返回到客户端
    /// seed 使用1轮随机数，未知
    export function createBox() {
        return {
            id: randomHex(),
            seed: randomNumber(),
        };
    }

    // box 为随机创建，key 为输入值
    function _openBox(box: LootBox, key: number) {
        // ts = 秒数
        const ts = Math.floor(Date.now() / 1000);
        switch (true) {
            // 1 相等
            case key === box.seed:
                return 1;
            // 2 秒数差小于 60
            /// 无关 seed
            case Math.abs(key - ts) < 60:
                return 2;
            // 3 seed 可被 key 除尽，且不得相等
            case box.seed % key === 0:
                return 3;
            // 4
            /// 无关 seed
            case Math.cos(key) * 0 !== 0:
                return 4;
            // 5 key 作为乘数使得与 seed 的积为 1337 的倍数
            /// 无关 seed：key = 1337 时一定为 1337 的倍数
            case key && (box.seed * key) % 1337 === 0:
                return 5;
            // 6 key 提取 seed 的低位后 `|` 保存原值
            case key && (box.seed | key) === (box.seed | 0):
                return 6;
            // -0 不小于 0，任意数 / -0 = 负无穷，负无穷 < 0
            /// 无关 seed
            case !(key < 0) && box.seed / key < 0:
                return 7;
            default:
                return 0;
        }
    }

    // 查看名称以映射分支
    export function openBox(box: LootBox, key: number) {
        console.log(`[BOX OPEN] seed=${box.seed} key=${key} result=${loot.name}`);
    }
    ```

- 查看随机函数 `src\utils\random.ts`
    ```js
    // 32位随机数
    export function randomNumber() {
        return Math.floor(Math.random() * 0xffffffff);
    }

    // 32位随机数 -> 16进制字符串 -> 0填充到至少8个字符
    // 拼接两轮随机数
    export function randomHex() {
        return (
            randomNumber().toString(16).padStart(8, "0") +
            randomNumber().toString(16).padStart(8, "0")
        );
    }
    ```

## 功能端点摘要

`/` 主页 
- 功能
    - 导航
        - `/about`
        - `/collection`
        - `/buy`

`/about`
- 功能
    - 文字描述

`/collection`
- 功能
    - 卡片式文字 + 图片

`/buy`
- 功能
    - 购买盒子，本端点 `/buy` POST
        - 参数
            - csrf token
                - 似乎反射 cookie 中的 csrf_token 
                - [ ] xss
                - [ ] 服务器端模板注入
            - `action=buy`

`/open` 
- 功能
    - 开启盒子 `/open` POST
        - 参数
            - csrf 令牌，仍与 cookie 相同
            - key
                - JS 脚本设置 key 的参数值为随机数
            - `boxId=2216bb7b9d54a909`
            - `action=open`
