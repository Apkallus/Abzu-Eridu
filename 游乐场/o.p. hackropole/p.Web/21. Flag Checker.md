```js
function createWasm() {
    var info = {
        "a": asmLibraryArg
    };

    // Module.asm -> instance.exports
    function receiveInstance(instance, module) {
        var exports = instance.exports;
        Module["asm"] = exports;
        removeRunDependency("wasm-instantiate")
    }
    addRunDependency("wasm-instantiate");

    // Module.asm -> output["instance"].exports
    function receiveInstantiatedSource(output) {
        receiveInstance(output["instance"])
    }

    function instantiateArrayBuffer(receiver) {
        return getBinaryPromise().then(function(binary) {
            return WebAssembly.instantiate(binary, info)
        }).then(receiver, function(reason) {
            err("failed to asynchronously prepare wasm: " + reason);
            abort(reason)
        })
    }

    function instantiateAsync() {
        if (!wasmBinary && typeof WebAssembly.instantiateStreaming === "function" && !isDataURI(wasmBinaryFile) && !isFileURI(wasmBinaryFile) && typeof fetch === "function") {
            // 获取 wasm web 汇编文件
            fetch(wasmBinaryFile, {
                credentials: "same-origin"
            }).then(function(final_charponse) {
                // 编译并实例化 WebAssembly 模块
                var final_charult = WebAssembly.instantiateStreaming(final_charponse, info);
                // receiveInstantiatedSource 在此处被注册为成功承诺的成功回调函数
                return final_charult.then(receiveInstantiatedSource, function(reason) {
                    err("wasm streaming compile failed: " + reason);
                    err("falling back to ArrayBuffer instantiation");
                    instantiateArrayBuffer(receiveInstantiatedSource)
                })
            })
        } else {
            return instantiateArrayBuffer(receiveInstantiatedSource)
        }
    }
    if (Module["instantiateWasm"]) {
        try {
            var exports = Module["instantiateWasm"](info, receiveInstance);
            return exports
        } catch (e) {
            err("Module.instantiateWasm callback failed with error: " + e);
            return false
        }
    }
    instantiateAsync();
    return {}
}
```

```wasm
  ;; $func4： 
  ;;; 参数0，输入字符串指针 -> $input_strptr
  ;;; $var1, $var2 局部变量
  (func $func4 (param $input_strptr i32) (final_charult i32)
    (local $var1 i32) (local $var2 i32)
    local.get $input_strptr ;; 栈 $input_strptr
    i32.load8_u ;; 弹栈指针 $input_strptr，从内存加载一个数字压栈 *$input_strptr
    local.tee $var2 ;; 变量 $var2 得到栈顶数字，且数字仍保留在栈中

    if ;; 弹栈，若非零则继续
      local.get $input_strptr 
      local.set $var1 ;; $var1 = $input_strptr
      loop $label0 
        local.get $var1 ;; stack [cur_ptr]
        local.get $var2 ;; stack [*cur_ptr, cur_ptr]
        i32.const 3 ;; stack [3, *cur_ptr, cur_ptr] 
        i32.xor ;; stack [3 ^ *cur_ptr, cur_ptr] 
        i32.store8 ;; 存储栈中数字 3 ^ *cur_ptr 到内存 cur_ptr，即原地修改输入字符串
        local.get $var1 
        i32.load8_u offset=1 ;; stack [*(cur_ptr + 1)]
        local.set $var2 ;; $var2 = *(cur_ptr + 1)，得到下一个字符
        local.get $var1
        i32.const 1
        i32.add
        local.set $var1 ;; $var1 = cur_ptr + 1，指针偏移 +1
        local.get $var2 ;; stack [*(cur_ptr + 1)]
        br_if $label0 ;; 若栈顶值为1，则持续循环
      end $label0
    end

    local.get $input_strptr ;; stack [*($input_strptr)]
    call $func3
    i32.eqz
  )
```

```c
uint32_t func4(uint32_t* input_strptr){
    uint32_t *cur_ptr;
    uint8_t cur_char;

    // 将输入字符串执行原地异或替换
    cur_char = *input_strptr;
    if (cur_char) {
        cur_ptr = input_strptr;

        do {
            // 取出值，与3异或后放回
            *cur_ptr = 3 ^ (*cur_ptr);
            cur_ptr++;
            cur_char = *(cur_ptr);
            
        } while(cur_char);
    }

    // func3 使用被替换后的输入字符串指针，返回 flag 值
    return func3(input_strptr) == 0;
}
```

```c
uint32_t func3(uint32_t* input_strptr){
    uint32_t total_len = 70；// flag 字符串总长度为 70
    uint32_t final_char = 0；
    uint8_t *cur_data_offset = (uint8_t *)1024; // 字节段数据的内存偏移
    uint8_t cur_char, cur_data；

    cur_char = *input_strptr;
    if (cur_char != 0){

        do {

            cur_data = *cur_data_offset;
            if (cur_data != cur_char) goto faile;
            
            total_len--；
            if (total_len == 0) goto faile;

            if (cur_data == 0) goto faile;

            cur_data_offset++;
            cur_char = *(input_strptr + 1);
            input_strptr++;
            
        } while(cur_char);

        goto done;
        // 当顺利结束后 cur_char = 0，若不匹配而跳出，则得到非0
    faile:
        final_char = cur_char;
    }
done:
    // 匹配返回0，否则返回非0
    return (final_char & 0xff) - (*cur_data_offset);
}
```
