    ```html
                    url: "/save",
                    data: $d
                })
                .done(function(data) {
                    r = $.parseJSON(data);
                    if (r.error) {
                        KO(r.error);
                    } else {
                        OK(r.id);
                    }
                })
        })

        function OK(id) {
            $('#status').html('<div id="alert" class="alert alert-success"><strong>Thanks!</strong> You can check the status of your feedback <a href="/status/' + id + '">here</a>.</div>')
            $('#status').show();
        }

        function KO(msg) {
            $('#status').html('<div id="alert" class="alert alert-warning"><strong>Error!</strong>' + msg + '</div>')
            $('#status').show();
        }

        $('#btn-clear').on('click', function(event) {
            event.preventDefault();
            $('#note').val('');
        });
    </script>
    ```
