# 实验室：通过客户端原型污染实现 DOM XSS

本实验室存在通过客户端原型污染导致的 DOM XSS 漏洞。要解决本实验：

1. 找到一个可用于向全局 `Object.prototype` 添加任意属性的源。
2. 识别一个允许执行任意 JavaScript 的小工具属性。
3. 结合这两者以调用 `alert()`。

你可以手动在浏览器中解决此实验，也可以使用 [DOM Invader](/burp/documentation/desktop/tools/dom-invader) 作为辅助。

        ```javascript
        // 为 jQuery 添加一个 URL 解析器，返回一个对象
        // 该函数旨在用于解析类似 window.location 的 URL
        // 用法：$.parseParams('http://mysite.com/?var=string') 或 $.parseParams() 解析当前 window.location
        // 简单变量：  ?var=abc                        返回 {var: "abc"}
        // 简单对象：  ?var.length=2&var.scope=123     返回 {var: {length: "2", scope: "123"}}
        // 简单数组：  ?var[]=0&var[]=9                返回 {var: ["0", "9"]}
        // 带索引的数组：?var[0]=0&var[1]=9            返回 {var: ["0", "9"]}
        // 嵌套对象：  ?my.var.is.here=5              返回 {my: {var: {is: {here: "5"}}}}
        // 混合使用：  ?var=a&my.var[]=b&my.cookie=no  返回 {var: "a", my: {var: ["b"], cookie: "no"}}
        // 注意：不支持在数组内嵌套对象，例如 ?var[1].test=abc 是无效的

        (function($) {
            var re = /([^&=]+)=?([^&]*)/g;
            var decode = function(str) {
                return decodeURIComponent(str.replace(/\+/g, ' '));
            };
            $.parseParams = function(query) {
                // recursive function to construct the result object
                function createElement(params, key, value) {
                    key = key + '';
                    // if the key is a property
                    if (key.indexOf('.') !== -1) {
                        // extract the first part with the name of the object
                        var list = key.split('.');
                        // the rest of the key
                        var new_key = key.split(/\.(.+)?/)[1];
                        // create the object if it doesnt exist
                        if (!params[list[0]])
                            params[list[0]] = {};
                        // if the key is not empty, create it in the object
                        if (new_key !== '') {
                            createElement(params[list[0]], new_key, value);
                        } else
                            console.warn('parseParams :: empty property in key "' + key + '"');
                    } else // if the key is an array
                    if (key.indexOf('[') !== -1) {
                        // extract the array name
                        var list = key.split('[');
                        key = list[0];
                        // extract the index of the array
                        var list = list[1].split(']');
                        var index = list[0]
                        // if index is empty, just push the value at the end of the array
                        if (index == '') {
                            if (!params)
                                params = {};
                            if (!params[key] || !$.isArray(params[key]))
                                params[key] = [];
                            params[key].push(value);
                        } else // add the value at the index (must be an integer)
                        {
                            if (!params)
                                params = {};
                            if (!params[key] || !$.isArray(params[key]))
                                params[key] = [];
                            params[key][parseInt(index)] = value;
                        }
                    } else // just normal key
                    {
                        if (!params)
                            params = {};
                        params[key] = value;
                    }
                }
                // be sure the query is a string
                query = query + '';
                if (query === '')
                    query = window.location + '';
                var params = {}, e;
                if (query) {
                    // remove # from end of query
                    if (query.indexOf('#') !== -1) {
                        query = query.substr(0, query.indexOf('#'));
                    }

                    // remove ? at the begining of the query
                    if (query.indexOf('?') !== -1) {
                        query = query.substr(query.indexOf('?') + 1, query.length);
                    } else
                        return {};
                    // empty parameters
                    if (query == '')
                        return {};
                    // execute a createElement on every key and value
                    while (e = re.exec(query)) {
                        var key = decode(e[1]);
                        var value = decode(e[2]);
                        createElement(params, key, value);
                    }
                }
                return params;
            }
            ;
        }
        )(jQuery);
        ```

            ```js
                if (new_key !== '') {
                    createElement(params[list[0]], new_key, value);
                }
            ```
