import sys
import os
import re
import json

def main():
    flag = os.environ.get("FLAG")

    if not flag:
        print("Flag was not read from environment. Aborting.")
        sys.exit(-1)
    else:
        flag_rand = re.search(r"{.*}$", flag)
        if flag_rand is None:
            print("Flag isn't wrapped by curly braces. Aborting.")
            sys.exit(-2)
        else:
            flag_rand = flag_rand.group()
            flag_rand = flag_rand[1:-1]

    new_flag = f"picoCTF{{wh47_c4n_u_d0_wPHP_{flag_rand}}}"
    flag_path = "/root/flag.txt"
    with open(flag_path, "w") as f:
        f.write(new_flag)

    # Create and update metadata.json =====================================

    metadata = {}
    metadata['flag'] = str(new_flag)
    json_metadata = json.dumps(metadata)

    with open("metadata.json", "w") as f:
        f.write(json_metadata)

    # =====================================================================

if __name__ == "__main__":
    main()
