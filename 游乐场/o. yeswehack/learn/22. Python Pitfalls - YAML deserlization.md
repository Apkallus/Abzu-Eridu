# Python 陷阱 - YAML 反序列化

Python 的函数 [`yaml.load`](https://docs.python.org/3/library/pathlib.html#pathlib.PurePath.joinpath) 用于将数据反序列化为 YAML。然而，如果被反序列化的 YAML 数据可由用户控制，且应用程序上的 YAML 加载器使用默认的 YAML 加载器，攻击者就可能在 YAML 反序列化过程中执行 Python 代码。

易受攻击的代码示例：

```python
import yaml
user_data = "!!python/object/apply:print ['pwned']"
data_yaml = yaml.load(user_data, Loader=yaml.Loader)
```

攻击者可以通过在 YAML 数据的开头添加 YAML 代码 `!!python/object/apply:print ['pwned']` 来利用 `yaml.load` 函数。这会指示 YAML `load` 函数执行 Python 代码 `print('pwned')`，从而使攻击者能够利用[代码注入 (CWE-94)](https://cwe.mitre.org/data/definitions/94.html)漏洞。

```py
import os, uuid, yaml, datetime
from urllib.parse import unquote
from jinja2 import Environment, FileSystemLoader
template = Environment(
    autoescape=True,
    loader=FileSystemLoader('/tmp/templates'),
)
os.chdir('/tmp')

def verifyPath(basePath, path):
    return basePath == os.path.dirname(os.path.join(path))

def main():
    # User input (game result filename)
    filename = unquote("§filename§")
    settings = unquote("§settings§")

    # Make dedicated user upload folder based on current session    
    userSession = uuid.uuid4()
    uploadPath = f"uploads/{userSession}/settings"
    os.makedirs(uploadPath, exist_ok=True)

    message = ""
    error = ""
    verbose = ""
    settingsYaml = None
    try:
        filePath = os.path.join(uploadPath, filename)
        
        if verifyPath(uploadPath, filePath) == False:
            raise Exception("invalid filename")

        settingsYaml = yaml.load(settings, Loader=yaml.Loader)

        # Write game result
        with open(filePath, "w") as f:
            f.write(settingsYaml)
        message = f"new settings has been added!"
    except Exception as e:
        if e is not None:
            error = f"Error: {e}"
        else:
            error = "Error: invalid settings"

    # /tmp/templates/index.html
    print( template.get_template('index.html').render(error=error) )

main()


```
