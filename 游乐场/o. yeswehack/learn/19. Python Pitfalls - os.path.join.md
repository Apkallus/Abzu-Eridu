# Python 陷阱 - os.path.join

Python 的函数 `os.path.join` 在将多个路径连接为单个路径时具有有趣的行为。如果我们在函数中的某处指定一个以绝对路径开头的绝对路径，例如 Linux 系统上的 `/etc/passwd`。我们的路径 `/etc/passwd` 将覆盖其后的所有路径并成为新的绝对路径。

脆弱代码示例：

```python
import os
payload = "/etc/passwd"
file = os.path.join("/user/uploads/", payload)
with open(file, "r") as f:
    print(f.read())  # 输出文件内容：“/etc/passwd”
```

import os, uuid, datetime
from urllib.parse import unquote
from jinja2 import Environment, FileSystemLoader
template = Environment(
    autoescape=True,
    # 此处设置模板加载目录
    loader=FileSystemLoader('/tmp/templates'),
)
os.chdir('/tmp')


def main():
    # User input (game result filename)
    filename = unquote("§输入 filename§")
    gameResult = unquote("§输入 result§")

    # Make dedicated user upload folder based on current session    
    userSession = uuid.uuid4()
    uploadPath = f"uploads/{userSession}/games"
    os.makedirs(uploadPath, exist_ok=True)

    message = ""
    error = ""
    try:
        filePath = os.path.join(uploadPath, filename)
        # Write game result
        with open(filePath, "w") as f:
            f.write(gameResult)
        message = f"game result uploaded to: {filePath}"
    except:
        error = "Error: no file was given"
        
    # 路径 /tmp/templates/index.html
    print( template.get_template('index.html').render(error=error, message=message) )

main()
