# Python 陷阱 - 类污染

与 JavaScript 中的 [原型污染](https://cheatsheetseries.owasp.org/cheatsheets/Prototype_Pollution_Prevention_Cheat_Sheet.html) 类似，当不可信输入修改类级别属性而非实例级别数据时，Python 可能会遭受 [类污染漏洞](https://blog.abdulrah33m.com/prototype-pollution-in-python/)。由于 Python 类是可变的，对类的单次写入会影响所有依赖该属性的现有和未来实例。如果攻击者能够影响诸如 `setattr`、反序列化对象或任何直接写入类的逻辑等函数，他们就可以改变全局行为。

易受攻击的代码示例：

```python
some_var = "change me!"

class Dummy:
    def __init__(self):
        pass

def merge(source, destination):
    for key, value in source.items():
        if hasattr(destination, "get"):
            if destination.get(key) and type(value) == dict:
                merge(value, destination.get(key))
            else:
                destination[key] = value
        elif hasattr(destination, key) and type(value) == dict:
            merge(value, getattr(destination, key))
        else:
            setattr(destination, key, value)

some_class = Dummy()

payload = {
    "__init__": {
        "__globals__": {
            "some_var": "polluted"
        }
    }
}

merge(payload, some_class)

print(some_var)  # 输出: polluted
```

```py
import os, json, uuid, datetime
from urllib.parse import unquote
from jinja2 import Environment, FileSystemLoader
template = Environment(
    autoescape=True,
    loader=FileSystemLoader('/tmp/templates'),
)
os.chdir('/tmp')

UPDATE_CMD = "sh update.sh"

class Player:
    def __init__(self, settings=None):
        for key, value in settings.items():
            setattr(self, key, value)

def merge(source, destination):
    for key, value in source.items():
        if hasattr(destination, "get"):
            if destination.get(key) and type(value) == dict:
                merge(value, destination.get(key))
            else:
                destination[key] = value
        elif hasattr(destination, key) and type(value) == dict:
            merge(value, getattr(destination, key))
        else:
            setattr(destination, key, value)

def update():
    os.popen(UPDATE_CMD).read()

def main():    
    player = Player({
        "username": "guest",
        "device": "keyboard",
        "difficulty": "medium",
        "autoSave": True
    })

    message = ""
    error = ""
    try:
        # Enable new player settings
        playerSettingsJSON = json.loads(unquote("§输入§"))
        merge(playerSettingsJSON, player)

        update()

        message = f"game settings updated!"

    except json.JSONDecodeError as err:
        error = f"Error: no settings was updated, error : {err}"
        
    # /tmp/templates/index.html
    print( template.get_template('index.html').render(error=error, message=message) )

main()


```
